🗃️ 数据结构

typedef struct {
    char name[32];      // 姓名
    char id[19];        // 身份证号
    char phone[12];     // 11位手机号
    int  call_count;    // 通话次数
} Customer;

// 开放寻址哈希槽
typedef struct {
    int valid;          // 0=空 1=占用 2=删除
    int cust_idx;       // customers下标
} OASlot;

// 链地址哈希节点
typedef struct ChainNode {
    int cust_idx;
    struct ChainNode *next;
} ChainNode;
